% =======================================
% ACEL'S MECHANISM 1
% =======================================
\begin{frame}{ACEL's DSGE Mechanism}
    ACEL builds two models on CEE JPE (2005):
    \begin{itemize}
        \item Homogeneous capital
        \item \textbf{Firm-specific capital}
    \end{itemize}
    \vspace{0.5cm}
    
    \begin{itemize}
        \item The key assumption is that, in any given period, a \textbf{aggregate capital stock is pre-determined and firm-specific}
        \item When firms re-optimise, \textbf{prices are adjusted by a small amount} because each firm's short-run marginal cost is increasing in its own output
        \item \textbf{Inflation is inertial} despite firms re-optimising prices more frequently than in the homogeneous capital model
    \end{itemize}    
\end{frame}

% =======================================
% ACEL'S MECHANISM 2
% =======================================

\begin{frame}{ACEL's DSGE Mechanism (Cont'd)}
\begin{itemize}
    \item When log-linearised, the only difference in the characterising equilibrium equations lies in the Phillips Curve
\end{itemize}

\begin{align}
    \Delta\hat{\pi}_t &= \mathbb{E} [\beta\Delta\hat{\pi}_{t+1} + \gamma\hat{s}_t|\Omega_t]
    \label{eq:Phillips_Curve}
    \\
    \gamma &= \frac{(1-\xi_p)(1-\beta\xi_p)}{\xi_p}\chi
    \label{eq:Phillips_Curve_Slope}
\end{align}

\begin{itemize}
    \item \textbf{Models are observationally equivalent in terms of prices and quantities:} \textcolor{PadovaRed}{macro data identify $\gamma$, the difference between the two models lies in the implied $\xi_p$}
    \item {\small $\hat{s}_t$ is the economy-wide marginal cost, $\gamma$ the Phillips Curve slope, $\xi_p$ the Calvo price stickiness parameter}
    \item {\small $\chi$ reduces the sensitivity of inflation to marginal costs in the firm-specific capital model}
\end{itemize}
\end{frame}

% =======================================
% ACEL'S MECHANISM 3
% =======================================

\begin{frame}{ACEL's DSGE Mechanism (Cont'd)}\label{slide:acel_dsge_mechanism}
 Here $\chi$ is a non-linear function representing the strategic complementarity in price setting 
\begin{itemize}
    \item In the \textbf{Homogeneous} model, $\chi \equiv 1$
    \item In the \textbf{Firm-specific} model, $\chi < 1$ and captures two forces that make a firm move its price \textit{less} when it can reset:
    \begin{itemize}
        \item It depends on capacity utilisation cost $\sigma_a$
        \\
              The costlier it is to vary capital utilisation, the steeper the firm's own marginal cost curve
        \item The more elastic the firm's demand, the larger the fall in output (and hence in marginal cost) after a price increase
    \end{itemize}
    \vspace{0.3cm}
    
    \item $\chi$ governs \textbf{how much} a firm moves its price
    \item $\xi_p$ \textbf{how often}, with implied duration $1/(1-\xi_p)$
\end{itemize}
 Given a fixed value of $\gamma$, a smaller $\chi$ implies a smaller $\xi_p$

    \end{center}

\end{frame}

% ILLUSTRATION OF THE PROBLEM: TRUNCATION BIAS 
% =======================================

\begin{frame}{Illustration of the problem: Truncation Bias}\label{slide:truncation_bias}
    \begin{equation}
        Y_t = B_1Y_{t-1} + \cdots + B_pY_{t-p} + u_t, \qquad \mathbb{E}(u_t u_t') = \Sigma
    \end{equation}

    \begin{equation}
        B(L) = B_1L + \cdots + B_pL^p
    \end{equation}

    \begin{itemize}
        \item The VAR with $p < p^{*}$ suffers from truncation bias.\footnote{Where $p^{*}$ denotes the DGP true lag length.}

        \item Lag truncation biases the reduced-form coefficients $B(L)$ and the reduced-form variance-covariance matrix $\Sigma$, which are the two main ingredients for IRF estimation.
    \end{itemize}

    \begin{equation}
        \mathrm{IRF} = f\bigl(B(L),\Sigma\bigr)
    \end{equation}

\begin{frame}{Long-Lag VARs}
    De Graeve and Westermark JME (2025) demonstrate that increasing the lag structure improves VAR IRF estimates, obtaining:
    \pause
    \vspace{0.4cm}
    
    \textbf{Lower Bias}
    \begin{itemize}
        \item By reducing the truncation bias misspecification at each horizon
    \end{itemize}
    \pause

    \vspace{0.4cm}
    \textbf{Lower Variance}
    \begin{itemize}
        \item Misspecification reduction lowers the variance of the residuals
        \item This variance reduction counteracts the imprecision caused by increased parameterisation
    \end{itemize}
\end{frame}



% =======================================
% LONG LAG 2 BENEFITS
% =======================================

\begin{frame}{Long-Lag VARs Benefits}
    The variance-reducing effect of lower misspecification often dominates the penalty of increased parameterisation.
    \vspace{0.4cm}
    
    Compared to short-lag VARs, Long-Lag VARs have:
    \begin{itemize}
        \item Reduced truncation bias
        \item More precise inference
        \item Lower MSE
        \item Better coverage
    \end{itemize}
\end{frame}

\begin{frame}{Conclusions}
    Following the advice of the Long-Lag VAR literature and replicating ACEL (2005) with an enriched VAR lag structure, I find that:
    \begin{itemize}
        \item Capacity Utilisation Cost, crucial variable to justify the firm-specific model, dramatically shrinks to 0
        \item Firm specificity of capital does not seem to be the key concept to explain inflation inertia
        \item ACEL's evidence on firm-specific capital is not robust to long-lag VARs
    \end{itemize} 
    \vspace{0.5cm}

        \item Long-Lag VARs might have structural implications; it's up to the researcher to demonstrate that their findings are robust to an enriched lag structure
    \end{itemize}

% ACEL'S Shock Propagation Mechanism: HOMOGENEOUS
% =======================================

\begin{frame}{ACEL's MP Shock Propagation Mechanism}\label{slide:acel_mp_shock_prop_mech}
\large

$\varepsilon_t^M\Down \Rightarrow i_t\Down$ $\Rightarrow$ $y^d_t\Up$
$\Rightarrow$ $y^s_t\Up$ $\Rightarrow h_t^d\Up$
\\
$\Rightarrow$ Calvo wages: $w_t\Up$, $\frac{w_t}{P_t}\Up$ little

\vspace{0.5cm}

\textbf{Homogeneous}
\begin{itemize}
    \setlength{\itemsep}{8pt}
    \item $K_t=u_t\bar K_t$ $\Rightarrow$ $K^d_t\Up$ but $\bar{K_t}$ and $u_t\Up$ $r_t\Up$ given $\hat{\sigma}_a$
    \item $\Rightarrow$ $MC_{i,t} = MC_t$ $\Rightarrow$ MC independent of $y_{i,t}$ $\chi=1$
    \item $MC_t\Up$ $\Rightarrow$ $P_t^{1-\xi_p}\Up$ $\Rightarrow$ $P_t\Up$ $\Rightarrow$ $\pi_t\Up$
    \item $\gamma = f(\overset{-}{\xi_p})$ given $\bar{\gamma}$ low, $\xi_p$ needs to be high enough
\end{itemize}

        \hyperlink{slide:acel_dsge_mechanism}{\beamerbutton{Back to ACEL's DSGE Mechanism}}
    \end{center}

\end{frame}

\begin{frame}{ACEL's MP Shock Propagation Mechanism (Cont'd)}
\large

$\varepsilon_t^M\Down \Rightarrow i_t\Down$ $\Rightarrow$ $y^d_t\Up$
$\Rightarrow$ $y^s_t\Up$ $\Rightarrow h_t^d\Up$
\\
$\Rightarrow$ Calvo wages: $w_t\Up$, $\frac{w_t}{P_t}\Up$ little

\vspace{0.5cm}
    
\textbf{Firm-Specific}
\begin{itemize}
    \setlength{\itemsep}{8pt}
    \item $K_{i,t}=u_{i,t}\bar K_{i,t}\Up$ $\Rightarrow$ $K^d_t\Up$ but $\bar{K_t}$
    \item Given $\chi < 1$ $\Rightarrow MC_{i,t}\Up$ in $y_{i,t}$
    \item $MC_{i,t}\Up$ $\Rightarrow$ $P_t^{1-\xi_p}\Up$ $\Rightarrow$ $P_t\Up$ $\Rightarrow$ $\pi_t\Up$
    \item $\gamma = f(\overset{-}{\xi_p};\overset{-}{\sigma_a})$ given $\bar{\gamma}$
\end{itemize}
\vspace{0.5cm}
\begin{itemize}
    \item If $\sigma_a \rightarrow 0$ $\Rightarrow$ $\chi \rightarrow 1$ $\Rightarrow$ $\gamma = f(\overset{-}{\xi_p};0)$ $\Rightarrow$ $\xi_{p,\text{FS}} \rightarrow \xi_{p,\text{H}}$
\end{itemize}

\begin{frame}{Identification Bias}\label{slide:identification_bias}
    Ravenna (2007) finds a second channel through which lag truncation operates: \textbf{\textcolor{PadovaRed}{Identification Bias}}
    \begin{itemize}
        \item It is a \textbf{different source of bias} from pure truncation
        \item It arises even when the identifying assumptions are correct, and it emerges in population
        \item \textbf{It appears when the AR coefficients enter the identifying map's computations of structural shocks}
        \item If the AR coefficients are biased, the identifying map is biased as well
        \vspace{0.3cm}
        
        \item It is not present in short-run Cholesky identification
        \item It is present in long-run restrictions identification à la Blanchard and Quah (1989)
    \end{itemize}
    
    \begin{center}
        \hyperlink{slide:truncation_bias}{\beamerbutton{Back to Truncation Bias}}
    \end{center}

\end{frame}

% =======================================
%  Long Rung Identification Fragilities
% =======================================
\begin{frame}{Long-Run Restrictions' Fragilities}
    Using the Blanchard and Quah (1989) identification as an example.
    \\
    It is a long-run Cholesky that restricts the cumulative effect of a shock on the level of a variable to be zero, which is equivalent to restricting the sum of the impulse responses to be zero.
    \vspace{0.3cm}
    \\
    A VAR($p$) does not estimate the sum consistently when the true representation is of infinite order,
    therefore the restrictions are imposed on the wrong object.
    \vspace{0.3cm}
    \\
    This has consequences for where the bias might appear. A long-run restriction says nothing 
    about any finite horizon. Therefore the bias can appear at any horizon, impact included, not just the long run.

    \begin{center}
        \hyperlink{slide:truncation_bias}{\beamerbutton{Back to Truncation Bias}}
    \end{center}

\end{frame}

% =======================================
\begin{frame}{Tech Shocks Comment}
    \begin{itemize}
        \item The Identification Bias is binding, and its effect can be seen in the fact that 
              the responses disagree across lags at every horizon
        \item Under short-run recursive identification, most impact responses to an MP shock are zero by construction.
              Pure truncation bias distorts the propagation of a correctly isolated shock, so it accumulates with the 
              horizon and cannot by itself generate a disagreement at $h = 0$
        \item MP shock responses have kept their sign and shape, moving only in timing
        \item Tech shock responses, identified by long-run restrictions, do not agree at any horizon, 
              including the impact, and the disagreement is not only in timing but also in sign
    \end{itemize}
