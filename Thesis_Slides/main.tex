% =======================================
% CEE and ACEL Motivation
% =======================================

\begin{frame}{CEE and ACEL's DSGE Motivation}
    \begin{itemize}
        \item Standard NK models replicate inflation persistence with levels of price stikiness, inconsistent with micro data evidence 
        \item CEE JPE (2005) and ACEL RoED (2011) seminal works introduced the concept of \textbf{firm specific capital} to model inflation persistence,
    mechanism supported by IRF matching with VAR(4)
    \end{itemize}
\end{frame}

% =======================================
% ACEL'S MECHANISM 1
% =======================================
\begin{frame}{ACEL's DSGE Mechanism}
        \begin{itemize}
        \item In Altig et al., (2005) inflation is inertial despite firms re-optimising prices on average every 1.5 quarters
        \item Inertia is explained by the fact that, when firms re-optimise, they do it by a small amount because each firm's short run marginal cost is increasing in its own output
        \item The key assumption is that, in any given period, a firm's capital stock is pre-determined and firm-specific
    \end{itemize}

    \vspace{0.5cm}
    They build two models on Christiano et al. JPE, (2005):
    \begin{itemize}
        \item Homogeneous capital
        \item Firm-specific capital
    \end{itemize}
\end{frame}

% =======================================
% ACEL'S MECHANISM 2
% =======================================

\begin{frame}{ACEL's DSGE Mechanism (Cont'd)}
\begin{itemize}
    \item When log-linearized, the only difference in the characterizing equilibrium equations lies in the Phillips Curve
\end{itemize}

\begin{align}
    \Delta\hat{\pi}_t &= \mathbb{E} [\beta\Delta\hat{\pi}_{t+1} + \gamma\hat{s}_t|\Omega_t]
    \label{eq:Phillips_Curve}
    \\
    \gamma &= \frac{(1-\xi_p)(1-\beta\xi_p)}{\xi_p}\chi
    \label{eq:Phillips_Curve_Slope}
\end{align}

\begin{itemize}
    \item $\hat{s}_t$ is the economy-wide marginal cost, $\gamma$ the Phillips Curve slope, $\xi_p$ the Calvo price stickiness parameter
    \item $\chi$ reduces the sensitivity of inflation to marginal costs in the firm-specific capital model
    \item The two models are observationally equivalent in terms of prices and quantities: macro data identify $\gamma$, so the difference between the two models lies in the implied $\xi_p$
\end{itemize}
\end{frame}

% =======================================
% ACEL'S MECHANISM 3
% =======================================

\begin{frame}{ACEL's DSGE Mechanism (Cont'd)}
\begin{itemize}
    \item In the \textbf{Homogeneous} model, $\chi = 1$: $\gamma$ depends only on $\beta$ and $\xi_p$
    \item In the \textbf{Firm-specific} model, $\chi < 1$ and captures two forces that make a firm move its price \textit{less} when it can reset:
    \begin{itemize}
        \item the costlier it is to vary capital utilization, the steeper the firm's own marginal cost curve
        \item the more elastic the firm's demand, the larger the fall in output (and hence in marginal cost) after a price increase
    \end{itemize}
    \vspace{0.3cm}
    \item So $\chi$ governs \textbf{how much} a firm moves its price
    \item $\xi_p$ \textbf{how often}, with implied duration $1/(1-\xi_p)$
\end{itemize}
 Since $\gamma$ is estimated by aggregate data, a smaller $\chi$ imply a smaller $\xi_p$
\end{frame}

% =======================================
% ACEL'S RESULTS
% =======================================

\begin{frame}{ACEL (2005) Results}
    The estimated $\gamma$ is 0.040, which implies:
    \begin{itemize}
        \item \textbf{Homogeneous} capital: $\xi_p = 0.82$, firms change prices every 5.6 quarters
        \item \textbf{Firm-specific} capital: firms change prices every 1.5 quarters
    \end{itemize}
    The gap arises because firms' demand curves are highly elastic and their marginal cost curves are very steep: the same $\gamma$ is delivered by frequent but small price adjustments.
\\
\vspace{0.1cm}
\textbf{Only the firm-specific capital model reconciles the micro-macro pricing conflict}
\vspace{0.5cm}

\end{frame}

\begin{frame}{DSGE, State Space and VAR($\infty$) Representation}
\begin{itemize}
    \item Many DSGE's log linearized solution can be mapped in State Space representation and then into VAR($\infty$)
    \item Standard lag choice by SVAR researchers is 4 for quarterly data and 12 for monthly
    \item If a DSGE is estimated via IRFME with a finite order VAR, the DSGE is estimate with a Biased by Truncation VAR
    
    \item VARs are typically small systems with few lags often interpret as models of the entire economy, IRFs are highly sensitive to variable and lag choice
\end{itemize}

\vspace{0.4cm} 

\pause
\begin{itemize}
    \item \textcolor{PadovaRed}{\textbf{Should the VAR lag order be increased to overcome Truncation Bias implications?}}
\end{itemize}

\end{frame}

% =======================================
% ILLUSTRATION OF THE PROBLEM: TRUNCATION BIAS 
% =======================================

\begin{frame}{Illustration of the problem: Truncation Bias}
    \begin{equation}
        Y_t = B_1Y_{t-1} + \cdots + B_pY_{t-p} + u_t, \qquad \mathbb{E}(u_t u_t') = \Sigma
    \end{equation}

    \begin{equation}
        B(L) = B_1L + \cdots + B_pL^p
    \end{equation}

    \begin{itemize}
        \item The VAR with $p < p^{*}$ suffers from truncation bias.\footnote{Where $p^{*}$ denotes the DGP true lag length.}

        \item Lag truncation biases the reduced form coefficients $B(L)$ and the reduced form variance covariance matrix $\Sigma$, which are the two main ingredients for IRF estimation.
    \end{itemize}

    \begin{equation}
        \mathrm{IRF} = f\bigl(B(L),\Sigma\bigr)
    \end{equation}
\end{frame}

% =======================================
% LONG LAG 1
% =======================================

\begin{frame}{Long-Lag VARs}
    De Graeve and Westermark, (2025) demontrates that increasing the lag structure improves VAR's IRFs estimates obtaining:
    \pause
    \vspace{0.4cm}
    
    \textbf{Lower Bias}
    \begin{itemize}
        \item By reducing the truncation bias misspecification at each horizon
    \end{itemize}
    \pause

    \vspace{0.4cm}
    \textbf{Lower Variance}
    \begin{itemize}
        \item Misspecification reduction lowers the variance of the residuals
        \item This variance reduction counteracts the imprecision caused by increased parametrisation
    \end{itemize}
\end{frame}



% =======================================
% LONG LAG 2 BENEFITS
% =======================================

\begin{frame}{Long-Lag VARs Benefits}
    The variance-reducing effect of lower misspecification often dominates the penalty of increased parameterisation.
    \vspace{0.4cm}
    
    Compared to short-lag VARs, Long-Lag VARs have:
    \begin{itemize}
        \item Reduced truncation bias
        \item More precise inference
        \item Lower MSE
        \item Better coverage
    \end{itemize}
\end{frame}

\begin{frame}{Conclusions}
    Following the advices from the Long-Lag VARs literature.
    \\
    Replicating ACEL et. al., (2005), enriching the VAR lag structure I find that:
    \begin{itemize}
        \item Capacity Utilization Cost, crucial variable to justify the firm specific model, dramatically shrinks to 0
        \item Firm specificity of capital seems not the key concept to explain inflation inertia
        \item ACEL's evidence on firm specific capital is not robust to long-lag VARs
        \vspace{0.5cm}
        \item Long-Lag VARs migth have structural implication, it's up to the researcher to demonstrate that its findings are robust to an encriched lag structure
    \end{itemize}
